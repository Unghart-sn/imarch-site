(function () {
  var form = document.getElementById('lp-collegium-form');
  if (!form) return;
  var status = document.getElementById('lp-form-status');
  var btn = form.querySelector('button[type="submit"]');
  var endpoint = form.getAttribute('data-endpoint') || '';
  function say(msg, cls) {
    status.textContent = msg;
    status.className = 'lp-form-status ' + (cls || '');
  }
  form.addEventListener('submit', function (e) {
    e.preventDefault();
    if (!form.reportValidity()) return;
    var payload = {};
    new FormData(form).forEach(function (v, k) { payload[k] = v; });
    if (!endpoint) {
      var body = 'Организация: ' + (payload.org || '—') +
                 '\nПочта: ' + payload.email +
                 '\n\n' + (payload.message || '');
      location.href = 'mailto:info@imarch.ltd?subject=' +
        encodeURIComponent('Коллегиум: запрос с сайта') +
        '&body=' + encodeURIComponent(body);
      return;
    }
    btn.disabled = true;
    say('Отправляем…', 'is-pending');
    fetch(endpoint, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(payload)
    })
      .then(function (r) { if (!r.ok) throw new Error('HTTP ' + r.status); return r; })
      .then(function () {
        form.reset();
        say('Запрос принят. Ответим на указанную почту.', 'is-ok');
      })
      .catch(function () {
        say('Не удалось отправить. Напишите на info@imarch.ltd', 'is-err');
      })
      .finally(function () { btn.disabled = false; });
  });
})();
</script>
